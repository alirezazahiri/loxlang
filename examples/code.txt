class Circle {
    init(radius) {
        this._radius = radius;
    }

    area {
        return 3.141592653 * this.radius * this.radius;
    }

    radius {
        return this._radius;
    }

    set radius(value) {
        this._radius = value;
    }
}

var circle = Circle(4);
print circle.area;

circle.radius = 2;
print circle.area;
